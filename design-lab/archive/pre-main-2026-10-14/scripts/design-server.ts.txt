import fs from 'node:fs/promises';
import crypto from 'node:crypto';
import {fileURLToPath} from 'node:url';
import type {Plugin} from 'vite';
export function designServer():Plugin {
 const token=crypto.randomBytes(24).toString('hex');
 const configPath=fileURLToPath(new URL('../src/config/design.json',import.meta.url));
 const pagePath=fileURLToPath(new URL('../src/authoring.html',import.meta.url));
 const reviewPath=fileURLToPath(new URL('../src/navigation-review.html',import.meta.url));
 const homeReviewPath=fileURLToPath(new URL('../src/home-review.html',import.meta.url));
 return {name:'local-design-desk',apply:'serve',configureServer(server){
  server.middlewares.use(async(req,res,next)=>{
   const path=req.url?.split('?')[0];
   if(path!=='/design/'&&path!=='/design/navigation/'&&path!=='/design/home/'&&path!=='/design/content/'&&path!=='/design/portrait/'&&path!=='/design/composition/'&&path!=='/design/material/'&&path!=='/design/emphasis/'&&path!=='/design/layout/'&&!path?.startsWith('/design/emphasis-proof/')&&!path?.startsWith('/design/portrait-proof/')&&path!=='/__design/config')return next();
   const host=new URL(`http://${req.headers.host||'invalid'}`).hostname;
   if(!['127.0.0.1','localhost','[::1]'].includes(host)){res.statusCode=403;return res.end('Local access only');}
   res.setHeader('Cache-Control','no-store');
   try {
    if(path==='/design/layout/'){
     if(req.method!=='GET'){res.statusCode=405;return res.end();}
     res.setHeader('Content-Type','text/html; charset=utf-8');
     return res.end(await server.transformIndexHtml(path,await fs.readFile(new URL('../src/layout-review.html',import.meta.url),'utf8')));
    }
    if(path==='/design/emphasis/'){
     if(req.method!=='GET'){res.statusCode=405;return res.end();}
     res.setHeader('Content-Type','text/html; charset=utf-8');
     return res.end(await server.transformIndexHtml(path,await fs.readFile(new URL('../src/emphasis-review.html',import.meta.url),'utf8')));
    }
    if(path?.startsWith('/design/emphasis-proof/')){
     const name=path.slice('/design/emphasis-proof/'.length);
     if(req.method!=='GET'||!/^(home|about|research)-(before|after)\.png$/.test(name)){res.statusCode=404;return res.end();}
     res.setHeader('Content-Type','image/png');
     return res.end(await fs.readFile(new URL('../../homepage-planning/browser-qa/body-emphasis/'+name,import.meta.url)));
    }
    if(path==='/design/material/'){
     if(req.method!=='GET'){res.statusCode=405;return res.end();}
     res.setHeader('Content-Type','text/html; charset=utf-8');
     return res.end(await server.transformIndexHtml(path,await fs.readFile(new URL('../src/material-review.html',import.meta.url),'utf8')));
    }
    if(path==='/design/composition/'){
     if(req.method!=='GET'){res.statusCode=405;return res.end();}
     res.setHeader('Content-Type','text/html; charset=utf-8');
     return res.end(await server.transformIndexHtml(path,await fs.readFile(new URL('../src/composition-review.html',import.meta.url),'utf8')));
    }
    if(path?.startsWith('/design/portrait-proof/')){
     const name=path.slice('/design/portrait-proof/'.length);
     if(req.method!=='GET'||!['01-tone.png','02-orange.png','02-blue.png','03-orange-screen.png','03-blue-screen.png','recipe.json'].includes(name)){res.statusCode=404;return res.end();}
     res.setHeader('Content-Type',name.endsWith('.png')?'image/png':'application/json');
     return res.end(await fs.readFile(new URL('../.design/portrait-plates/'+name,import.meta.url)));
    }
    if(path==='/design/portrait/'){
     if(req.method!=='GET'){res.statusCode=405;return res.end();}
     res.setHeader('Content-Type','text/html; charset=utf-8');
     return res.end(await server.transformIndexHtml(path,await fs.readFile(new URL('../src/portrait-review.html',import.meta.url),'utf8')));
    }

    if(path==='/design/content/'){
     if(req.method!=='GET'){res.statusCode=405;return res.end();}
     // This standalone review is loaded outside Astro's page graph. Refresh its
     // data dependencies so a reload cannot show copy from an older SSR cache.
     for(const source of ['content','publications','provenance']){
      const file=fileURLToPath(new URL(`../src/data/${source}.ts`,import.meta.url));
      for(const mod of server.moduleGraph.getModulesByFile(file)||[])server.moduleGraph.invalidateModule(mod);
     }
     const {contentReview}=await server.ssrLoadModule('/src/design/content-review.ts');
     res.setHeader('Content-Type','text/html; charset=utf-8');return res.end(contentReview());
    }
    if(path==='/design/'||path==='/design/navigation/'||path==='/design/home/'){
     if(req.method!=='GET'){res.statusCode=405;return res.end();}
     const html=(await fs.readFile(path==='/design/'?pagePath:path==='/design/home/'?homeReviewPath:reviewPath,'utf8')).replace('__DESIGN_TOKEN__',token);
     res.setHeader('Content-Type','text/html; charset=utf-8');return res.end(await server.transformIndexHtml(path,html));
    }
    res.setHeader('Content-Type','application/json');
    if(req.method==='GET')return res.end(await fs.readFile(configPath,'utf8'));
    if(req.method!=='POST'){res.statusCode=405;return res.end('{}');}
    const expected=`http://${req.headers.host}`;
    if(req.headers.origin!==expected||req.headers['x-design-token']!==token||!req.headers['content-type']?.startsWith('application/json')){res.statusCode=403;return res.end(JSON.stringify({error:'仅允许本地设计台保存配置'}));}
    let body='';
    for await(const chunk of req){body+=chunk.toString();if(Buffer.byteLength(body)>65536){res.statusCode=413;return res.end(JSON.stringify({error:'配置过大'}));}}
    // Load through Vite so edits to parameter definitions invalidate the validator too.
    const {validateConfig}=await server.ssrLoadModule('/src/design/settings.ts');
    const candidate=JSON.parse(body);validateConfig(candidate);
    const temp=configPath+'.'+crypto.randomBytes(6).toString('hex')+'.pending';
    await fs.writeFile(temp,JSON.stringify(candidate,null,2)+'\n');await fs.rename(temp,configPath);
    return res.end(JSON.stringify({saved:true}));
   }catch(error){res.statusCode=400;res.end(JSON.stringify({error:error instanceof Error?error.message:'无法保存配置'}));}
  });
 }};
}
